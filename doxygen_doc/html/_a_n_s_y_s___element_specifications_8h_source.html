36/**
37
38@brief Represents ANSYS element type specifications that are needed
39prior to model construction in order to identify corresponding
40element types in CSMP++.
41
42@author S.K. Matthai
43@date 2003
44
45
46@section motivation Motivation
47
48Relates the ANSYS finite element types to the generic finite element
49names that are used inside of CSP. Also additional diagnostics are
50obtained as to whether the element is a line, surface, or volume, and
51whether the element uses linear, quadratic or cubic basis functions.
52Furthermore the element can be queried for the number of neighbors
53and its minimum spatial dimension.
54
55
56@section implementation Implementation
57
58ANSYS definitions of element types and their identification
59numbers are the following. They are only partially mapped to
60CSP and incorporated into enumerations to avoid the pollution of
61the csp namespace.
62
63@code

104#define EDGE_ELEMENT_MASK ((1 << BAR_2) | (1 << BAR_3))
105#define SURFACE_ELEMENT_MASK ((1 << TRI_3) | (1 << TRI_6) | \

110 (1 << PYRA_5) | (1 << HEXA_27) | \
111 (1 << PENTA_18) | (1 << PYRA_14) | \
112 (1 << PYRA_13))
113
114
115#define LINEAR_ELEMENT_MASK ((1 << NODE) | (1 << BAR_2) | \
116 (1 << TETRA_4) | (1 << HEXA_8) | \
117 (1 << TRI_3) | (1 << PENTA_6) | \
118 (1 << QUAD_4) | (1 << PYRA_5))
119
120#define QUADRATIC_ELEMENT_MASK ((1 << BAR_3) | (1 << TETRA_10) | \
121 (1 << HEXA_20) | (1 << TRI_6) | \
122 (1 << PENTA_15) | (1 << QUAD_8) | \
123 (1 << QUAD_9) | (1 << HEXA_27) | \
124 (1 << PENTA_18) | (1 << PYRA_14) | \
125 (1 << PYRA_13))
126@endcode
