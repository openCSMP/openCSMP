113 const double fNorm(1.0 / std::sqrt(dotProduct(vectorToNormalize, vectorToNormalize)));

116 [fNorm](const double val){ return val * fNorm; } );

151 /**
152
153 Calculates the center of equal masses at the vertices of a planar, closed,
154 convex surface defined by vecPoints.
155
156 @section arguments Input Arguments
157
158 @param vecPoints The points representing the vertices.
159
160 @param vecCentroid the centroid of the point cloud.
161
162 @section implementation Implementation
163
164 At first all the points are summed, and then each of these summed components
165 is divided by the total number of vertices.
166
167 @section application Application
168
169 The main idea is to calculate the center of mass delimited by the given
170 vertices, supposing that the body is of constant density.
171 */
172 template<uint32_t dim>
173 inline void vertexCenterOfMass( const std::vector<Point<dim> >& vecPoints, Point<dim>& vecCentroid )
174 {
175 for( size_t i{0U}; i < vecPoints.size(); i++)
176 vecCentroid += vecPoints[i];
177
178 vecCentroid /= vecPoints.size();
179 }
