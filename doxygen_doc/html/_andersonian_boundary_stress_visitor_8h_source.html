25/**
26 @brief Apply stresses to the boundaries of a three-dimensional geomechanic model.
27
28 @author SKM 29/9/2014
29
30 The stresses are entered via a StressRegime object, but the variable 'overburden pressure'
31 storing the vertical stress Sv must be defined and initialised.
32
33 The AndersonianBoundaryStressVisitor accesses all Boundary objects,
34 and resolves the boundary normal stress and shear stresses using the
35 normals of the boundary Face objects which must be outward pointing.
36
37 In the corresponding calculations body forces must be enabled.
38
39 @note If a boundary is inclined or vertical, this is fine. In this case,
40 the visitor takes the vertical variation in the boundary normal stress into account.
41
42 Input variables are the stress state supplied by the StressRegime object.
43
44 The visitor also computes the Face variables 'normal stress' and 'shear stress'
45 on the visited boundaries. The nodal forces that it will apply
46 are computed from these.
47
48 @attention by default all forces that were previously assigned are
49 zeroed first.
50
51 @attention Confining stresses are always compressive and result inward pointing forces.
52
53 @attention this visitor assumes that the far-field stress regime is
54 Andersonian, i.e. one of the principal stresses is oriented perpendicular
55 to the earth surface, i.e. that one of the principal stress axes is aligned
56 with the vertical (Y)-axis of the CSMP model.
57
58 @note This visitor allows for a workaround for applying stress boundary conditions
59 if one does not want to use the Neumann condition functionality of the PDE_Integrator.
60*/
61class AndersonianBoundaryStressVisitor : public Visitor<3U> {
62 public:
63 /// by default, any entries into the RHS force vectors will get overwritten
64 AndersonianBoundaryStressVisitor( const Model<3U>&,
65 const StressRegime&,
66 const char* SV_variable="overburden pressure", ///< should vary with depth
67 bool overwrite_force_vector=false );
68
69 virtual ~AndersonianBoundaryStressVisitor();
70
71 /// to zero out force vector if overwrite_force_vector=true, prior to accumulation of forces
72 virtual void Visit( Boundary<3U>* );
73
74 /// computes nodal forces ensuing from stress and weighted by number of nodes of the boundary face
75 virtual void Visit( Face<3U>* );
76
77 private:
78 const StressRegime& stress_regime_; ///< input Andersonian stress state
79 StressRotate Andersonian_stress_; ///< the full tensor constructed for the stress regimes
80
81 const csmp::Index Sv_key_, ///< vertical (overburden) stress (a scalar)
82 Sn_key_, ///< normal stress acting on boundary faces
83 Ss_key_; ///< shear stress acting on boundary faces
84 const csmp::Index F_key_; ///< (nodal) 'force' (N)
85 VectorVariable<3U> vc_, force_; ///< forces read from the nodes
86 VectorVariable<3U> nrml_; ///< outward-pointing unit normal to face
87 TensorVariable<3U> far_field_stress_; ///< as computed from stress regime
88 ScalarVariable Sv_, Sn_, Ss_; ///< scalar vertical, normal and shear stresses
89 bool overwrite_previous_forces_;
90};
91
92} // end csmp
