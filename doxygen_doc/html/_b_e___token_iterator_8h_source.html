31template<class InputIter, class Pred = Isalpha>
32class TokenIterator {
33 InputIter first;
34 InputIter last;
35 std::string word;
36 Pred predicate;
37public:
38 TokenIterator(InputIter begin, InputIter end,
39 Pred pred = Pred())
40 : first(begin), last(end), predicate(pred) {
41 ++*this;
42 }
43 TokenIterator() {} // End sentinel
44 // Prefix increment:
45 TokenIterator& operator++() {
46 word.resize(0);
47 first = std::find_if(first, last, predicate);
48 while(first != last && predicate(*first))
49 word += *first++;
50 return *this;
51 }
52 // Postfix increment
53 class CaptureState {
54 std::string word;
55 public:
56 CaptureState(const std::string& w) : word(w) {}
57 std::string operator*() { return word; }
58 };
59 CaptureState operator++(int) {
60 CaptureState d(word);
61 ++*this;
62 return d;
63 }
64 // Produce the actual value:
65 std::string operator*() const { return word; }
66 const std::string* operator->() const { return &word; }
67 // Compare iterators:
68 bool operator==(const TokenIterator&) {
69 return word.size() == 0 && first == last;
70 }
71 bool operator!=(const TokenIterator& rv) {
72 return !(*this == rv);
73 }
74};
