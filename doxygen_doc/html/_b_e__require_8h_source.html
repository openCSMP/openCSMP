17inline void require(bool requirement,
18 const std::string& msg = "Requirement failed"){
19 using namespace std;
20 if (!requirement) {
21 fputs(msg.c_str(), stderr);
22 fputs("\n", stderr);
23 exit(1);
24 }
25}

27inline void requireArgs(int argc, int args,
28 const std::string& msg =
29 "Must use %d arguments") {
30 using namespace std;
31 if (argc != args + 1) {
32 fprintf(stderr, msg.c_str(), args);
33 fputs("\n", stderr);
34 exit(1);
35 }
36}
37
38inline void requireMinArgs(int argc, int minArgs,
39 const std::string& msg =
40 "Must use at least %d arguments") {
41 using namespace std;
42 if(argc < minArgs + 1) {
43 fprintf(stderr, msg.c_str(), minArgs);
44 fputs("\n", stderr);
45 exit(1);
46 }
47}

49inline void assure(std::ifstream& in,
50 const std::string& filename = "") {
51 using namespace std;
52 if(!in) {
53 fprintf(stderr, "Could not open file %s\n",
54 filename.c_str());
55 exit(1);
56 }
57}

59inline void assure(std::ofstream& out,
