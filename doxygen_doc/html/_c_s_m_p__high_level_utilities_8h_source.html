38/// scientific-format string with appropriate digits (without loss of precision (std::to_string limits to 6 significant digits)
39template <typename T>
40inline std::string numberToString( const T& value, bool verbose=false ) {
41 if constexpr( std::is_same<T,double>::value ) {
42 // int snprintf ( char * s, size_t n, const char * format, ... );
43 // sign + digit + dp + digits + e + sign + expo + \0 (15 significant digits)
44 std::string dstring( 1 + 1 + 1 + (DBL_DECIMAL_DIG - 1) + 1 + 1 + 5, '\0');
45 auto slen = snprintf( &dstring[0], dstring.size(), "%1.14e", value );
46 dstring.resize(slen);
47 if ( verbose ) std::cout <<"\nmain: floating point value: "<< dstring <<" vs. "<< value << std::endl;
48 return dstring;
49 }
50 else if constexpr ( std::is_same<T,float>::value )
51 {
52 // sign + digit + dp + digits + e + sign + expo + \0 (8 significant digits)
53 std::string fstring( 1 + 1 + 1 + (FLT_DECIMAL_DIG - 1) + 1 + 1 + 3, '\0');
54 auto slen = snprintf( &fstring[0], fstring.size(), "%1.7e", value );
55 fstring.resize(slen);
56 if ( verbose ) std::cout <<"\nmain: floating point value: "<< fstring <<" vs. "<< value << std::endl;
57 return fstring;
58 }
59 return number_to_string( value );
60}

79/**
80@brief Vector element subtraction.
81
82@author P. Lang
83@date 06/11/2012
84
85@param [in,out] toRemoveFrom vector from which elements are deleted.
86@param [in,out] toRemove vector which contains elements to be deleted from toRemoveFrom
87
88@return new size of vector.
89
90*/
91template<typename T>
92size_t removeVectorElements( std::vector<T>& toRemoveFrom, std::vector<T>& toRemove )
93{
94 std::sort( toRemoveFrom.begin(), toRemoveFrom.end() );
95 std::sort( toRemove.begin(), toRemove.end() );
96 std::vector<T> toRemoveFromNew;
97 set_difference( toRemoveFrom.begin(), toRemoveFrom.end(), toRemove.begin(), toRemove.end(), std::back_inserter( toRemoveFromNew ) );
98 const size_t deleted( toRemoveFrom.size() - toRemoveFromNew.size() );
99 toRemoveFrom.swap( toRemoveFromNew );
100 std::vector<T> cache( toRemoveFrom );
101 toRemoveFrom.swap( cache );
102 return deleted;
103}
