116 struct vec3 {
117 double x, y, z;
118
119 vec3( double x0, double y0, double z0){
120 x = x0; y = y0; z = z0;
121 }
122
123 vec3 operator-( vec3 p ) const {
124 return vec3{ x - p.x, y - p.y, z - p.z };
125 }
126
127 vec3 cross( vec3 p ) const {
128 return vec3{
129 y * p.z - p.y * z,
130 z * p.x - p.z * x,
131 x * p.y - p.x * y
132 };
133 }
134
135 double dot( vec3 p ) const {
136 return x * p.x + y * p.y + z * p.z;
137 }
138
139 double norm() const {
140 return std::sqrt( x*x + y*y + z*z );
141 }
142 };
143
144 struct face3 {
145 std::vector<vec3> v;
146
147 vec3 normal() const {
148 assert( v.size() > 2 );
149 vec3 dir1 = v[1] - v[0];
150 vec3 dir2 = v[2] - v[0];
151 vec3 n = dir1.cross( dir2 );
152 double d = n.norm();
153 return vec3{ n.x / d, n.y / d, n.z / d };
154 }
155 };

158 bool is_inside_polyheadron( vec3 const& cp, std::vector<vec3> const& pts )
159 {
160 std::vector<face3> polyhedron{ // faces with 4 points
161 face3{ { pts[7], pts[6], pts[2], pts[3] } }, // front
162 face3{ { pts[4], pts[0], pts[1], pts[5] } }, // back
163 face3{ { pts[7], pts[3], pts[0], pts[4] } }, // left
164 face3{ { pts[6], pts[5], pts[1], pts[2] } }, // right
165 face3{ { pts[3], pts[2], pts[1], pts[0] } }, // top
166 face3{ { pts[7], pts[4], pts[5], pts[6] } }, // bottom
167 };
168
169 for ( face3 const& f : polyhedron ) {
170 vec3 p2f = f.v[0] - cp; // an arbitrary point on face
171 double d = p2f.dot( f.normal() );
172 d /= p2f.norm();
173
174 double bound = -1e-15; // to exclude boundaries
175 if ( d < bound )
176 return false;
177 }
178
179 return true;
180 }

182 size_t generateCellCentroid(ColumnCell& cell) {
183 Point<3> pts[8], cp(0, 0, 0), tcp(0, 0, 0), bcp(0, 0, 0);
184
185 pts[0] = p0->GetPoint(cell.z[0][0]);
186 tcp += pts[0];
187 pts[1] = p1->GetPoint( cell.z[1][0] );
188 tcp += pts[1];
189 pts[2] = p2->GetPoint( cell.z[2][0] );
190 tcp += pts[2];
191 pts[3] = p3->GetPoint( cell.z[3][0] );
192 tcp += pts[3];
193 pts[4] = p0->GetPoint(cell.z[0][1]);
194 bcp += pts[4];
195 pts[5] = p1->GetPoint(cell.z[1][1]);
196 bcp += pts[5];
197 pts[6] = p2->GetPoint(cell.z[2][1]);
198 bcp += pts[6];
199 pts[7] = p3->GetPoint(cell.z[3][1]);
200 bcp += pts[7];
201
202 tcp = tcp * 0.25;
203 bcp = bcp * 0.25;
204 cp = (tcp + bcp) * 0.5; // in normal cases
205 Point<3> rcp0 = (pts[0] + pts[6])*0.5; // in exceptial cases such as skewed cells
206 Point<3> rcp1 = (pts[2] + pts[4])*0.5;
207 Point<3> rcp2 = (pts[1] + pts[7])*0.5;
208 Point<3> rcp3 = (pts[3] + pts[5])*0.5;
209
210 std::vector<vec3> cp_vecs;
211 cp_vecs.push_back( vec3( cp[0], cp[1], cp[2] ));
212 cp_vecs.push_back( vec3( rcp0[0], rcp0[1], rcp0[2] ) );
213 cp_vecs.push_back( vec3( rcp1[0], rcp1[1], rcp1[2] ) );
214 cp_vecs.push_back( vec3( rcp2[0], rcp2[1], rcp2[2] ) );
215 cp_vecs.push_back( vec3( rcp3[0], rcp3[1], rcp3[2] ) );
216
217 std::vector<vec3> vecs; vecs.reserve( 8 );
218 for(auto p: pts) vecs.push_back( vec3( p[0], p[1], p[2] ) );
219
220 // find a valid centroid which is inside of its cell
221 bool found (false);
222 for ( auto v : cp_vecs ) {
223 if ( is_inside_polyheadron( v, vecs ) ){
224 cp = Point<3>( v.x, v.y, v.z );
225 found = true;
226 break;
227 }
228 }
229
230 // the centroid is not inside of the cell, return zero, otherwise return its new node index
231 if ( found ) {
232 size_t idx = ordinaryNodes.size() + extraNodes.size();
233 extraNodes.push_back( cp );
234 return idx;
235 }
236 else
237 return found;
238 }

271 bool ConstructHexahedron(ColumnCell& cell) {
272 //std::cerr << "hexa: \n";
273 for ( uint32_t i = 0; i < 8; ++i) {
274 vertexIDs[i] = i;
275 auto p = getNodeCoord(cell, vertexIDs[i]);
276 grid.ConvertFromReservoirToCSMPcoordinateSystem(p);
277 hexa.XYZ(i, 0, p[0]);
278 hexa.XYZ(i, 1, p[1]);
279 hexa.XYZ(i, 2, p[2]);
280 //std::cerr << "p" << i << " = " << p[0] << ' ' << p[1] << ' ' << p[2] << '\n';
281 }
282
283 uint32_t iNrIps = hexa.IntegrationPoints();
284 for (uint32_t iIp = 0u; iIp < iNrIps; ++iIp) {
285 hexa.JacobianAtIntegrationPoint(iIp);
286 double jacdet = hexa.JacobianDeterminant();
287 //std::cerr << "jacdet = " << jacdet << '\n';
288 if (jacdet <= 0) {
289 return false;
290 }
291 }
292 return true;
