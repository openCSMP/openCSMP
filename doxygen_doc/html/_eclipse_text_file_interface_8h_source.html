175template<class STREAM>
176bool EclipseTextFileInterface<STREAM>::ReadFile( std::ifstream& ifs,
177 STREAM* stream,
178 bool (STREAM::*IsCommentLineFunction)(char*),
179 bool (STREAM::*ReadFunction)( std::ifstream&, char*, size_t, std::set<std::string>&, std::string&, std::vector<std::string>& ),
180 bool fail_for_unknown_keyword )
181{
182 csmp::ErrorHandler& error_handler( csmp::ErrorHandler::Instance() );
183
184 // Read block data marked by keywords
185 do{
186 // read non-empty text line
187 readNonBlankLine( ifs, text_line_, line_length_ );
188 if ( ifs.eof() )
189 {
190 if( error_handler.Verbose() )
191 std::cout <<"\nEclipseTextFileInterfaceInClass<dim>::ReadFile: Reading completed!" << std::endl;
192 ifs.close();
193 return true;
194 }
195 // analyze text line
196 if ( !(stream->*IsCommentLineFunction)(text_line_) )
197 {
198 // read keyword and it's parameters
199 readKeyword( ifs, text_line_, line_length_,
200 keyword_, keyword_parameters_ );
201 // read first line in block if not blank
202 csmp::readNonBlankLine( ifs, text_line_, line_length_ );
203 if ( ifs.eof() )
204 {
205 if( error_handler.Verbose() )
206 std::cout <<"\nEclipseTextFileInterfaceInClass<dim>::ReadFile: Reading completed!" << std::endl;
207 ifs.close();
208 return true;
209 }
210 // check the keyword
211 if( fail_for_unknown_keyword && keywords_.find( keyword_) == keywords_.end() )
212 {
213 std::cerr<< "EclipseTextFileInterfaceInClass<dim>::ReadFile: Can not read data in block marked by keyword: "<< keyword_ << std::endl;
214 std::cerr<< "Please correct the keyword that you've used in your file" << std::endl;
215 std::cerr<< "The list of the avaliable keywords: "<< std::endl;
216 size_t i(1);
217 for( std::set<std::string>::const_iterator it = keywords_.begin(); it != keywords_.end(); it++, i++)
218 std::cerr<<"Keyword [ "<<i<< " ] = "<<(*it)<<std::endl;
219 std::cerr <<"\nEclipseTextFileInterfaceInClass<dim>::ReadFile: Reading was done with errors!" << std::endl;
220 ifs.close();
221 error_handler.Note( csmp::FATAL_ERROR,
222 "EclipseTextFileInterfaceInClass<dim>::ReadFile:",
223 "Undefined keyword:",
224 keyword_ );
225 return false;
226 }
227 // read data
228 else if( !(stream->*ReadFunction)( ifs, text_line_, line_length_,
229 keywords_, keyword_, keyword_parameters_ ) )
230 {
231 std::cerr <<"\nEclipseTextFileInterfaceInClass<dim>::ReadFile: Reading was done with errors!" << std::endl;
232 ifs.close();
233 error_handler.Note( csmp::FATAL_ERROR,
234 "EclipseTextFileInterfaceInClass<dim>::ReadFile:",
235 "Can not read data in block marked by keyword: ",
236 keyword_ );
237 return false;
238 }
239 }
240 }
241 while ( !ifs.eof() );
242
243 if( error_handler.Verbose() )
244 std::cout <<"\nEclipseTextFileInterfaceInClass<dim>::ReadFile: Reading completed!" << std::endl;
245
246 ifs.close();
247
248 return true;
249
250} // end ReadInputFile
