23/**
24 Line "face" that knows its surface Face neighbors.
25 Supports computations on the boundaries of 3D models.
26 Variable placement is FACE.
27 Can be a manifold.
28
29 @todo design issue: edges should only exist only in 3D models because, only there, they can have Face neighbors.
30 @todo remove templatisation, making this a 3D only entity
31*/
32template<uint32_t dim>
33class Edge : public FiniteElementPolicy<dim,Edge>,
34 public FiniteVolumePolicy<dim,Edge>,
35 public LocalVariableStorage<dim,Edge> {
36 public:
37 /// custom constructor, finds face ids and assigns nodes automatically
38 Edge( FiniteElement*,
39 const FiniteVolumeStencil<dim>*,
40 Face<dim>* const inner_parent,
41 Face<dim>* const outer_parent, ///< if any
42 const LocalVariables&,
43 const IntegrationPointVariables& );
44
45 ~Edge() {}
46
47 /// Local variable storage interface
48 PLACEMENT Placement() const { return FACE; }
49
50 uint32_t Nodes() const { return static_cast<uint32_t>(node_connector_.size()); }
51 uint32_t Neighbors() const { return 2U; }
52 uint32_t ConnectedNeighbors() const;
53
54 /// sides of Face object by analogy with Element
55 uint32_t Faces() const { return 2U; }
56
57 /// only constant iterators are provided because the user is not supposed to change the node pr neighbor connectivity (done by MeshManager)
58 typename std::vector<Node<dim>*>::const_iterator NodesBegin() const;
59 typename std::vector<Node<dim>*>::const_iterator NodesEnd() const;
60
61 /// to apply visitors whose application level is Boundary and target is Edge
62 void Accept( Visitor<dim>& );
63
64 /// access the nodes that are connected to the Edge
65 Node<dim>* const N( uint32_t n_local ) const;
66
67 /// access the neighbor faces of this face
68 Edge<dim>* const Neighbor( uint32_t ) const;
69
70 /// initialisation of nodes x dim matrix with the coordinates of the Edge's nodes
71 void NodeCoordinateMatrix( DenseMatrix<DM_MIN>& ) const;
72
73 /// on-the-fly 0..n-1 numbering stored in a mutable local variable (therefore const)
74 void Idx( size_t ) const;
75 size_t Idx() const;
76
77 /// higher-dimensional Face located on side opposite to where the unit normal points; will always be present
78 Face<dim>* const InnerParent() const;
79
80 /// higher-dimensional Face located on the side of the face to which the unit normal points; @attention does not exist on model boundary
81 Face<dim>* const OuterParent() const;
82
83 Face<dim>* const Parent( INTERFACE_SIDE ) const;
84
85 /// returns which Face of the higher dimensional inner neighbor Face this Edge shares its nodes with
86 uint32_t InnerParentFaceID() const;
87 uint32_t OuterParentFaceID() const;
88
89 // ------------------------------------------------------------------------
90 // Face geometry operations
91 // ------------------------------------------------------------------------
92
93 /// returns length of the Edge; method assumes same role as Volume() for the element
94 double Length() const;
95
96 /// the centre of gravity of the element
97 Point<dim> BaryCenter() const;
98
99 void Out() const;
100
101 private:
102 Face<dim>* innerParent_; ///< higher-dimensional neighbor in opposite direction of unit normal (always there)
103 Face<dim>* outerParent_; ///< higher-dimensional neighbor element in direction of interface normal
104 std::vector<Node<dim>*> node_connector_; ///< pointers to the nodes of the face
105 mutable size_t idx_;
106 uint32_t inner_parent_face_id_ = UNSPECIFIED; ///< face number of inside higher-dimensional parent Face
107 uint32_t outer_parent_face_id_ = UNSPECIFIED; ///< face number of outside higher-dimensional parent Face
108 Edge<dim>* nbor_edge0_=nullptr;
109 Edge<dim>* nbor_edge1_=nullptr; ///< the neighbors of the edge
110};
