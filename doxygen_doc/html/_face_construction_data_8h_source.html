17/**
18 @brief Container of pairs of pointers to Element objects juxtaposed along a shared face identified by its local face number.
19 Used in the construction of Boundary and SplitBoundary objects.
20
21 The juxtaposed elements must have the same dimension as the model.
22 The first element in the pair is the inside one and the second one the outside one.
23 Material values identify which regions they belong to.
24 There is also a Face or InterFace patch number used to identify the new modelsubdomain that will be created with the help
25 of FaceConstructionData.
26
27 @attention FaceConstructionData is populated by the function
28
29 higherDimensionalNeighbors( csmp::Element<dim>&, const csmp::Index& );
30*/
31template<uint32_t dim>
32class FaceConstructionData {
33 public:
34 FaceConstructionData( Element<dim>& parent_element,
35 std::pair<Element<dim>*,Element<dim>*>& neighbor_elements,
36 std::pair<uint32_t,uint32_t>& neighbor_element_faces,
37 std::pair<long,long>& neighbor_element_materials,
38 long this_material );
39
40 FaceConstructionData( Element<dim>& parent_element,
41 std::pair<Element<dim>*,Element<dim>*>& neighbor_elements,
42 std::pair<uint32_t,uint32_t>& neighbor_element_faces,
43 std::pair<long,long>& neighbor_element_materials,
44 long this_material,
45 uint32_t number_of_patch_this_data_belongs_to );
46
47 // NO ASSIGMENT OPERATOR BECAUSE ALL CLASS MEMBERS ARE CONSTANT
48
49 /// inner (first) and out (second) higher-order parent element
50 std::pair<Element<dim>*,Element<dim>*> NeighborElements() const;
51
52 /// idx of lower-dimensional element from which face data were derived if it exists
53 Element<dim>* LowerDimElement();
54
55 /// idx of element on the opposite site of the outward pointing normal
56 Element<dim>* const InnerElement() const;
57
58 /// the local id of the face located at the boundary
59 uint32_t InnerElementFace() const;
60
61 /// idx of element on the side to which the normal points to
62 Element<dim>* const OuterElement() const;
63 uint32_t OuterElementFace() const;
64
65 /// the material out of which the element consists from which the boundary face shall be constructed
66 long ElementMaterial() const;
67
68 /// in the order: 1) parent, 2) inner, 3) outer
69 void Materials( std::vector<long>& ) const;
70
71 /// Returns the materials of the neighboring higher-dimensional elements, inside followed by outside
72 std::pair<long,long> Materials() const;
73
74 /// identifier of the boundary that the face will belong to
75 void PatchNumber( uint32_t number );
76 uint32_t PatchNumber() const;
77
78 void Out() const;
79
80 private:
81 FaceConstructionData();
82
83 private:
84 Element<dim>& dim_m1_element_; ///< idx of lower-dimensional parent element
85 const std::pair<Element<dim>*,Element<dim>*> neighbors_; ///< indices of higher-dimensional elements on inside (first) and outside (second)
86 const std::pair<uint32_t,uint32_t> nbor_faces_; ///< matching faces on inside (first) and outside (second)
87 const std::pair<long,long> materials_; ///< integer codified juxtaposed regions
88 const long material_; ///< of the element from which the Face shall be constructed
89 uint32_t patch_number_; ///< unique identifier for the internal boundary patch this data relates to
90};
