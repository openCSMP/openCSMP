49inline double GenericTransferFunction::krw_BC( double sw_eff ) const
50 {
51 if ( sw_eff <= 0. ) return static_cast<double>(0.);
52 if ( sw_eff >= 1. ) return static_cast<double>(1.);
53
54 return std::pow( sw_eff, (2. + 3. * BC_param_) / BC_param_ );
55 }

58inline double GenericTransferFunction::krn_BC( double sw_eff ) const
59 {
60 if ( sw_eff <= 0. ) return static_cast<double>(1.);
61 if ( sw_eff >= 1. ) return static_cast<double>(0.);
62
63 return ((1. - sw_eff) * (1. - sw_eff)) * (1. - std::pow( sw_eff, (2. + BC_param_) / BC_param_ ));
64 }
