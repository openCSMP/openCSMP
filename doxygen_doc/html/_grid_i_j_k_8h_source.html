35 /// activitate or deactivate cells

37 assert( i < grid_dim_I_ && j < grid_dim_J_ && k < grid_dim_K_ );
38 return cell_activity_[i + j * grid_dim_I_ + k * grid_dim_I_ * grid_dim_J_ ];
39 }
