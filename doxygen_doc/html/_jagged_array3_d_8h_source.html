20/** custom hash for std::tuple<int, int, int>
21
22 The STL uses just uses the function
23 @code
24 std::size_t operator()(const std::tuple<int, int, int>& t) const {
25 std::size_t h1 = std::hash<int>{}(std::get<0>(t)); // Hash of the first element
26 std::size_t h2 = std::hash<int>{}(std::get<1>(t)); // Hash of the second element
27 std::size_t h3 = std::hash<int>{}(std::get<2>(t)); // Hash of the third element
28 // Combine the hashes using XOR and bit-shifting
29 return h1 ^ (h2 << 1) ^ (h3 << 2);
30 }
31 @endcode
32*/
33struct TupleHash {
34 std::size_t operator()( const std::tuple<uint32_t, uint32_t, uint32_t>& t ) const {
35 std::size_t h1 = std::hash<uint32_t>{}(std::get<0>(t));
36 std::size_t h2 = std::hash<uint32_t>{}(std::get<1>(t));
37 std::size_t h3 = std::hash<uint32_t>{}(std::get<2>(t));
38 // combine the hashes using XOR and bit-shifting
39 return h1 ^ (h2 << 1) ^ (h3 << 2);
40 }
41};

44/**
45 Three-dimensional jagged array (where the number of columns can vary), implemented via an unordered_map plus hashing function 'TupleHash'
46 Tested and compared in 'JaggedArray3D_Comparison_Test'.
47 For dense regularly shaped matrix, JaggedArray3D is slower than DynamicArray3D and a vector of vecto of vectors.
48 Good uses are for sparse 3D matrices the entries of which are in a flux (changing rapidly during a computation).
49*/
50template<class T>
51class JaggedArray3D {
52 public:
53 using key = std::tuple<uint32_t, uint32_t, uint32_t>;
54 using umap = std::unordered_map<key,T,TupleHash>;
55 // ref
56 using reference = typename std::unordered_map<key,T>::reference;
57 using const_reference = typename std::unordered_map<key,T>::const_reference;
58 // iter
59 using iterator = typename std::unordered_map<key,T>::iterator;
60 using const_iterator = typename std::unordered_map<key,T>::const_iterator;
61
62 /// default constructor constructs an empty array
63 JaggedArray3D() = default;
64
65 /// construction using initialiser list (which must be known at compile time)
66 explicit JaggedArray3D( std::initializer_list<std::initializer_list<std::initializer_list<T>>> );
67
68 /// construct from a monster vector of vector of vectors
69 explicit JaggedArray3D( const std::vector<std::vector<std::vector<T>>>& );
70
71 // ACCESSORS
72
73 /// iterators
74 iterator begin() { return data_.begin(); }
75 iterator end() { return data_.end(); }
76 const_iterator begin() const { return data_.begin(); }
77 const_iterator end() const { return data_.end(); }
78
79 /// Insert or modify elements
80 void set( uint32_t k, uint32_t i, uint32_t j, T value ) {
81 data_[{k, i, j}] = value;
82 }
83
84 /// read elements using operator() return NaN if they don't exist
85 T operator()( uint32_t k, uint32_t i, uint32_t j ) const {
86 key index = {k, i, j};
87 auto it = data_.find(index);
88 if (it != data_.end()) {
89 return it->second; // Element exists, return its value
90 } else {
91 return std::numeric_limits<T>::quiet_NaN(); // Element does not exist
92 }
93 }
94
95 /// read/modify elements (slower)
96 T& operator()( uint32_t k, uint32_t i, uint32_t j ) {
97 const key index = {k, i, j};
98 return data_[ index ];
99 }
100
101 /// quickest way to test for element existance with a predefined tuple (use std::make_tuple)
102 bool contains( const std::tuple<uint32_t, uint32_t, uint32_t>& index ) {
103 if ( data_.find(index) != data_.end() ) return true;
104 return false;
105 }
106
107 /// erase an element at a specific location
108 void erase( uint32_t k, uint32_t i, uint32_t j ) {
109 key index = {k, i, j};
110 // key index = std::make_tuple(k, i, j ); // slower
111 // check if the key exists in the map
112 auto it = data_.find(index);
113 if (it != data_.end()) {
114 // key found, erase the element
115 data_.erase(it);
116 } else {
117 // Key not found, throw an exception or silently ignore
118 throw std::out_of_range("JaggedArray3D::erase:: array element does not exist at the specified indices.");
119 }
120 }
121
122 // DIAGNOSTICS
123 uint32_t slices() const;
124
125 uint32_t rows() const;
126
127 /// max because the number of columns can vary in a jagged array
128 uint32_t max_cols() const;
129
130 /// returns the numbers of slices, rows, and maximum number of columns
131 std::tuple<uint32_t, uint32_t, uint32_t> max_kij() const;
132
133 /// print all elements to std out
134 void Out() const {
135 for (const auto& [index, value] : data_) {
136 auto [k, i, j] = index;
137 std::cout << "Value at (" << k << ", " << i << ", " << j << ") = " << value << std::endl;
138 }
139 }
140
141 private:
142 umap data_;
143};
