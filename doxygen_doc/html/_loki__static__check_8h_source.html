36////////////////////////////////////////////////////////////////////////////////
37// macro STATIC_CHECK
38// Invocation: STATIC_CHECK(expr, id)
39// where:
40// expr is a compile-time integral or pointer expression
41// id is a C++ identifier that does not need to be defined
42// If expr is zero, id will appear in a compile-time error message.
43////////////////////////////////////////////////////////////////////////////////
44
45#define STATIC_CHECK(expr, msg)
46 { Loki::CompileTimeError<((expr) != 0)> ERROR_##msg; (void)ERROR_##msg; }
47
