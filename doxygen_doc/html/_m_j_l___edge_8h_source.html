177/**
178
179@return SKEW_CROSS if the line segments intersect (in constrast with the infinite lines)
180COLLINEAR, PARALLEL
181SKEW_NO_CROSS as appropriate
182t = x value of intersection if intersection occurs
183*/
184inline INTERSECTION Edge::Cross( const Edge& e, double& t ) const
185 {
186 double s;
187 INTERSECTION crossType = e.Intersect( *this, s );
188 if ((crossType == COLLINEAR || crossType == PARALLEL)) return crossType;
189 if ((s < 0.) || (s > 1.)) return SKEW_NO_CROSS;
190 Intersect( e, t );
191 if ((0. <= t) && (t <= 1.)) return SKEW_CROSS;
192 return SKEW_NO_CROSS;
193 }
