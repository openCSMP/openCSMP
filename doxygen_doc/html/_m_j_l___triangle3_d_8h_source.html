62inline int Triangle3D::Classify( const Point3D& p ) const
63 {
64 Point3D v = p - v0_;
65 double len = v.Length();
66 if ( len == 0. ) return ON;
67 v = (1. / len) * v;
68 double d = v.DotProduct(n_);
69 if ( d > EPSILON1) return POSITIVE;
70 else if ( d < -EPSILON1) return NEGATIVE;
71 return ON;
72 }
