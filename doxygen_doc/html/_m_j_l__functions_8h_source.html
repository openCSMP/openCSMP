60/**
61
62@warning polygon must have been built in counterclockwise way (positive) for
63this to work !
64
65advances the current edge of polygon A clockwise and inserts this edge's
66(destination) endpoint x into intersection polygon R in counter-clockwise
67fashion, if x of A is inside R and x was not the last point inserted into
68R.
69*/
70inline void advance( const Polygon& A, Polygon& R, bool inside )
71 {
72 // advancing A clockwise (polygon was built counter-clockwise)
73 A.Advance(CLOCKWISE);
74
75 // add to new polygon in counter-clockwise fashion
76 if ( inside && (R.Point() != A.Point()) ) R.Insert( A.Point() );
77 }
