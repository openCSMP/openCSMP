23/**
24 Regional properties shared by multiple elements, but have variations
25 that can be calculated knowing other local properties discretized
26 on the elements.
27
28 Example: Young's modulus can be calculated from the value stored
29 here with corrections based on the deviation of the local porosity
30 from that for which E was determined.
31
32 Such calculations must be informed by petrophysical correlations
33 and the results might have to be perturbed by some randon variable
34 to reflect natural variations.
35
36*/
37struct MechanicalProperties {
38 /// granite properties as default
39 MechanicalProperties();
40
41 /// from property values supplied as a vector
42 explicit MechanicalProperties( const std::vector<double>& properties );
43
44 /// partial initialisation
45 MechanicalProperties( double E, double K, double nu,
46 double tensile_strength, double fric_coeff );
47
48 double E_; ///< Young's modulus
49 double B_; ///< bulk modulus = compressibility of rock skeleton = K_dry
50 double G_; ///< shear modulus = modulus of rigidity
51 double alpha_; ///< Biot coefficient alpha = 1 - K_dry/K_grain
52 double TS_; ///< tensile strength
53 double UCS_; ///< unconfined compressive strength
54 double pstar_; ///< crushing pressure of the rock, see Fjaer et al. 08', p. 68
55 double mu_; ///< friction coefficient
56 double nu_; ///< Poisson's ratio
57 double C_; ///< cohesive strength = inherent shear strength
58
59 // TODO: here we need the correlations that relate the properties to variations in the parent rock type
60 // derived properties for fault rocks: dilatation, compaction etc.
61
62 void Out() const;
63};

65} // end namespace csmp
