20/**
21 Contiguous mesh patch that can, but does not have to be integrated into a Model.
22 The idea is to temporarily store parts of a mesh during rebuilding operations;
23 Size info is used also to keep number information uptodate.
24
25 @author SKM
26 @date 23/5/21
27*/
28template<uint32_t dim>
29class MeshPatch {
30 public:
31 /// specify whether this is a volume, surface or line element patch
32 explicit MeshPatch( CELL_SHAPE cs ) : cell_dimension_{cs} {}
33
34/// creates an interconnected element patch from the shared faces of the supplied elements; replicates nodes upon request, and returns perimeter nodes and size of patch
35 size_t BuildInterveningPatch( const std::vector<std::pair<std::pair<Element<dim>*,uint32_t>,
36 std::pair<Element<dim>*,uint32_t> > >&,
37 bool copy_nodes,
38 std::vector<Node<dim>*> perimeter_node_ptrs );
39
40 /// how many different finite element types the patch contains
41 size_t CellTypes() const;
42 /// whether this is a LINE, SURFACE or VOLUME mesh
43 CELL_SHAPE CellGeometry() const;
44 size_t Cells() const;
45 size_t Nodes() const;
46
47 typename plf::colony<Element<dim>>::iterator ElementsBegin();
48 typename plf::colony<Element<dim>>::iterator ElementsEnd();
49
50 typename plf::colony<Node<dim>>::iterator NodesBegin();
51 typename plf::colony<Node<dim>>::iterator NodesEnd();
52
53 private:
54 std::map<CSMP_FEM_TYPE,FiniteElement*> fe_ptrs_; ///< finite-element pointers used in the FE policy
55 plf::colony<Node<dim>> nodes_; ///< nodes (if they are pre-existing, a light copy without properties is made)
56 plf::colony<Element<dim>> elements_; ///< elements
57 CELL_SHAPE cell_dimension_; ///< types of elements in the patch
58};

60} // end csmp
