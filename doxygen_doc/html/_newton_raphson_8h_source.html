29/// \brief This template class is an encapsulation of the one dimensional NewtonRaphson iteration algorithm. It allows for calculating df twice at each iteration, or d2f, to increase the spead of convergance. It also can be used in situations with boundaries. The orignal source cand be viewed or downloaded <a href="NewtonRaphson.h">here.</a><br/>

110 /// \brief this returns the number of times the bounaries have been hit. If boundaries are hit more than twice, then the loop is stoped.
111 int check_boundary( real & x, real df_dx) {
112
113 if (_check_boundary) {
114 _lower_boundary_hit = _upper_boundary_hit = false;
115 if (x < _min_x) {
116 x = _min_x + 2 * _epsilon / df_dx;
117 _lower_boundary_hit = true;
118 _number_boundary_hits++;
119 } else if (x > _max_x) {
120 x = _max_x - 2 * _epsilon / df_dx;
121 _upper_boundary_hit = true;
122 _number_boundary_hits++;
123 }
124 _boundary_hit = _lower_boundary_hit || _upper_boundary_hit;
125 }
126
127 return _number_boundary_hits;
128 }

203 /// This function does the calculations.
204 void do_iteration( ostream* output) {
205 int i;
206
207 _converged = _diverged = _boundary_solution = false;
208
209 _x[0] = _x0;
210
211 for (i=_number_boundary_hits=0; ;) {
212 _F_x[i] = (_F.*_f)( _x[i]) - _fsolve;
213
214 _abs_F_x[i] = (_F_x[i] > 0)? _F_x[i] : -_F_x[i];
215
216 if (_abs_F_x[i] < _epsilon) {
217 _x_converge = _x[i];
218 _f_converge = _F_x[i];
219 _converged = true;
220 break;
221 }
222
223 if (i>0 && _abs_F_x[i] > _abs_F_x[i-1]) {
224 _diverged = true;
225 break;
226 }
227
228 if (output) {
229 *output << i << " " << _x[i] << " " << _F_x[i] << " " << _dF_x[i] << endl;
230 }
231
232 _dF_x[i] = (_F.*_df)( _x[i]);
233
234 ++i;
235 if (i >= _max_iter) break;
236
237 if (_d2f != 0) {
238 real dx = (_F_x[i-1]) / _dF_x[i-1];
239 real df2_dx2 = (_F.*_d2f) ( _x[i-1]);
240 _dF_x[i-1] -= .5 * dx * df2_dx2;
241 } else if (_twice_df) {
242 _x[i] = _x[i-1] - (_F_x[i-1]) / _dF_x[i-1];
243 if (check_boundary( _x[i], _dF_x[i-1])>2) {
244 goto boundary;
245 }
246
247 if (_boundary_hit) {
248 _dF_x[i-1] = (_F.*_df)( _x[i]);
249 } else {
250 real df_dx = (_F.*_df)( _x[i]);
251 _dF_x[i-1] = .5 * (_dF_x[i-1] + df_dx);
252 }
253 }
254
255 _x[i] = _x[i-1] - (_F_x[i-1]) / _dF_x[i-1];
256
257 if (check_boundary( _x[i], _dF_x[i-1])>2) {
258 goto boundary;
259 }
260
261 continue;
262
263boundary:
264 if (_lower_boundary_hit) {
265 _boundary_solution = true;
266 _x_converge = _min_x;
267 _f_converge = (_F.*_f)( _min_x);
268 break;
269 }
270
271 if (_upper_boundary_hit) {
272 _boundary_solution = true;
273 _x_converge = _max_x;
274 _f_converge = (_F.*_f)( _max_x);
275 break;
276 }
277 }
278
279 if (_diverged && output) {
280 *output << "answer diverged" << endl;
281 }
282
283 if (_boundary_solution && output) {
284 *output << i << " " << _x_converge << " " << _f_converge << " boundary_solution" << endl;
285 } else if (_converged && output) {
286 *output << i << " " << _x[i] << " " << _F_x[i] << " answer converged" << endl;
287 }
288 }
