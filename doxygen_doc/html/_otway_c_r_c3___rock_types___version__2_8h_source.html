22/**
23 Using the "tuple" container of different (rock) types, a mapping is created between the rocktype numbers sampled from Otway CRC3 and
24 the names of the rock so that they can be seen during debugging.
25
26 The properties of the rocktypes are hard-coded into the corresponding classes CRC3_RockType1 to 15.
27
28 The HeterogeneityAndRateAwareModel gets the rock types from the 'rocktype' variable and choses the corresponding sets of curve fits.
29
30 For any of the composite rocktypes there are lamination parallel and perpendicular saturation functions. These are averaged dependent on the flow direction.
31
32 For composites, only the drainage curves are dependend on flow rate, the imbibition ones are not.
33
34 The imbibition curves are at the capillary limit.
35
36 Some of the composites are treated as mixtures between impermeable nonporous carbonate cemented rocks
37 and single rock types. In these cases conventional relperms are used for the remaining porous subtypes,
38 together with a downscaled porosity, and an anisotropic permeability.
39 These are:
40
41 M_CbSst_Mst Massive bedding (Carbonate-cemented sandstone lamina and mudstone matrix) - ONLY ONE SUBTYPE FLOWS
42 M_CbSst_Slt Massive bedding (Carbonate-cemented sandstone lamina and siltstone matrix) - ONLY ONE SUBTYPE FLOWS
43 M_CbSst_FSs Massive bedding (Carbonate-cemented sandstone lamina and fine sandstone matrix) - ONLY ONE SUBTYPE FLOWS
44 M_CbSst_CSst Massive bedding (Carbonate-cemented sandstone lamina and coarse sandstone matrix) - ONLY ONE SUBTYPE FLOWS
45
46 No flow will occur through:
47
48 H_CbSst, // -- Homogeneous (Carbonate-cemented sandstone) -- NO_FLOW
49
50 which leaves the 6 composites:
51
52 P_Mst_Slt, 4 COMPOSITE -- Planar bedding (Mudstone lamina and siltstone matrix)
53 P_Mst_FSst, 6 COMPOSITE: Planar bedding (Mudstone lamina and fine sandstone matrix)
54 P_Mst_CSst, // 8 COMPOSITE: Planar bedding (Mudstone lamina and coarse sandstone matrix)
55 P_Slt_FSst, // 11 COMPOSITE: MAARTJE SAMPLE# - planar bedding (Siltstone lamina with fine sandstone matrix)
56 P_Slt_CSst, // 12 COMPOSITE: Planar bedding (Siltstone lamina with coarse sandstone matrix)
57 X_CSst_FSst, // 14 COMPOSITE: Cross bedding (Fine sandstone lamina and coarse sandstone matrix)
58
59 Then there are 4 rocktypes for which we can use standard VG / BC relperms:
60
61 H_Mst, // -- Homogeneous Mudstone
62 H_Slt, // -- Homogeneous Siltstone
63 H_FSst, // -- Homogeneous (Fine Sandstone)
64 H_CSst // -- 15 Homogeneous (Coarse sandstone)
65
66 Those composite rocktypes where one of the components is carbonate cemented, can treated as single types with anisotropic perm and high irreducible sw
67
68 @attention the averaging is not done inside of this class.
69
70 Extra rock types to define the laminations within the composites?
71
72*/
73enum class RockName : int8_t
74 {
75 // from Achyut & Kuncho's file (July 2019), mostly composites
76 WELL, // 0 by default
77 H_Mst, // 1 Homogeneous Mudstone
78 M_CbSst_Mst, // 2 Massive bedding (Carbonate-cemented sandstone lamina and mudstone matrix) - ONLY ONE SUBTYPE FLOWS
79 H_CbSst, // 3 Homogeneous (Carbonate-cemented sandstone) -- NO_FLOW
80 P_Mst_Slt, // 4 COMPOSITE -- Planar bedding (Mudstone lamina and siltstone matrix)
81 M_CbSst_Slt, // 5 Massive bedding (Carbonate-cemented sandstone lamina and siltstone matrix) - ONLY ONE SUBTYPE FLOWS

116// UTILITY FUNCTIONS
117// functions used repeatedly in the curve fitting for layer-parallel and perpendicular flows that are produced by Maartje
118
119/// r_w used in Chierici model (AbdAllah addition)
120inline double r_w( double Sw, double Swr ) {
121 assert(Sw != 1.);
122 return (Sw - Swr) / (1. - Sw);
123 }
124
125/// gamma used in Chierici model (AbdAllah addition)
126inline double gamma( double ux, double a, double b, double c, double d ) {
127 return d + (a / (1. + b * std::pow( ux, c )));
128 }
129
130/// Wetting phase relative permeability (Chierici model) (AbdAllah addition)
131inline double k_w( double rw, double gamma_W_1, double gamma_W_2 ) {
132 return std::exp( -1. * gamma_W_1 * std::pow( rw , -1. * gamma_W_2) );
133 }
134
135/// Non-wetting phase relative permeability (Chierici model) (AbdAllah addition)
136inline double k_nw( double rw, double gamma_NW_1, double gamma_NW_2 ) {
137 return std::exp( -1. * gamma_NW_1 * std::pow( rw , gamma_NW_2) );
138 }

141
142/// effective wetting phase saturation
143inline double seff( double Sw, double Swr ) {
144 return (Sw - Swr) / (1. - Swr);
145 }
146
147/// effective wetting phase saturation taking into account residual saturation of the gas phase
148inline double seff( double Sw, double Swr, double Sgr ) {
149 return (Sw - Swr) / (1. - Swr - Sgr);
150 }
151
152/// effective wetting phase saturation, limited to range 0..1
153inline double seffL( double Sw, double Swr ) {
154 return std::max( std::min( (Sw - Swr) / (1. - Swr), 1. ), 0. );
155 }
156
157/// effective wetting phase saturation taking into account residual saturation of the gas phase, limited to range 0..1
158inline double seffL( double Sw, double Swr, double Sgr ) {
159 return std::max( std::min( (Sw - Swr) / (1. - Swr - Sgr), 1. ), 0. );
160 }
161
162/// repetitive square product involving effective water saturation
163inline double seffL_Product( double Sw, double Swr ) {
164 const double Seff = seffL(Sw,Swr);
165 return (1. - Seff * Seff) * (1 - Seff) * (1 - Seff);
166 }
167
168/// repetitive square product involving effective water saturation
169inline double seffL_Product( double Sw, double Swr, double Sgr ) {
170 const double Seff = seffL(Sw,Swr,Sgr);
171 return (1. - Seff * Seff) * (1 - Seff) * (1 - Seff);
172 }
173
174/// capillary limit Van Genuchten for wetting phase; the exponents are averaged
175inline double krw_VG( double Sw, double m_ave ) {
176 const double t1 = std::sqrt(Sw);
177 const double t3 = std::pow(Sw, 0.1e1 / m_ave);
178 const double t5 = std::pow(0.1e1 - t3, m_ave);
179 const double t7 = std::pow(0.1e1 - t5, 0.2e1);
180 return t7 * t1;
181 }
182
183/// Brooks-Corey, capillary limit approximation
184inline double krn_BC( double Sw ) {
185 const double temp = 1. - Sw * Sw;
186 return temp * temp;
187 }
188
189
190/// Brooks-Corey lambda parameter from VG m parameter, Lenhard et al. (1989)
191inline double lambdaFrom_VG( double m ) {
192 assert( m > 0. );
193 // Lenhard model does not work; used curve fit instead
194 // const double Sweff_ref(0.8);
195 // return (m / (1. - m)) * (1. - std::pow( std::min(Sweff_ref,0.99), 1./m ));
196 return 0.5e-1 * std::exp(5.8 * m) + 1.;
197 }

200/// log10 slope for logarithmic extension of pc following Webb (2000), Pc0(sw=0)=PC_MAX, Pc_star is Pc(Sw_star) = tangent poin where extension meets standard pc curve
201inline double pc_Slope( double Pc0, double Pc_star, double Sw_star ) {
202 const double t1 = log10(Pc0);
203 const double t2 = std::log10(Pc_star);
204 return -0.1e1 / Sw_star * (t1 - t2);
205 }
206
207
208/// logarithmic extension of PC following Webb (2000); Sw=actual water saturation, Pc0(sw=0)=PC_MAX, Pc_star is Pc(Sw_star) = tangent poin where log extension starts
209inline double pc_LogExtension( double Sw, double Sw_star, double Pc_star ) {
210 const double MAX_CAPILLARY_PRESSURE(4e7),
211 slope_pc(pc_Slope(MAX_CAPILLARY_PRESSURE,Pc_star,Sw_star)),
212 t3 = std::log10(Pc_star);
213 return std::pow(0.10e2, slope_pc * (Sw - Sw_star) + t3);
214 }

216/**
217 Brooks-Corey capillary pressure correlation for drainage of a water wet medium.
218
219 Piecewise definition over entire saturation range:
220 BC function down to effective water saturation, Seff= 1%, linear slope below Seff=0.01.
221*/
222inline double pc_BC( double Sw, double swr, double pd, double bcp ) {
223 assert( bcp > 0. );
224 assert( Sw >= 0. );
225 const double Seff = seff(Sw,swr);
226 if ( Seff >= 0.01 )
227 return pd * std::pow( Seff, -1. / bcp );
228
229 // linear extension from Sw to Sw=0 at pc_max
230 const double pc01 = pd * std::pow( 0.01, -1. / bcp ),
231 sw_star = -0.01 * swr + 0.01 + swr;
232
233 return pc_LogExtension( Sw, sw_star, pc01 );
234 }

238/// pc(sw) model, Maartje 20/12/19
239inline double pc_VG( double Sw, double pd, double m, double Swi_pc ) {
240
241 const double Sw_star = seff(Sw,Swi_pc), Pc_MAX(1.0e+7);
242 const double t2 = std::pow(Sw_star, -0.1e1 / m);
243 const double t5 = std::pow(t2 - 0.1e1, 0.1e1 - m);
244 return std::min( t5 * pd, Pc_MAX );
245
246 }

250/// curve fitting polynomial with 2 coefficients used by Maartje for water relperms
251inline double polyC2( double Sw, double C1, double C2 ) {
252 if ( Sw > 1. ) return 1.;
253 return std::sqrt(Sw) * (1. - std::pow( 1. - std::pow(Sw,C1), C2));
254 }
255
256/// curve fitting polynomial with 3 coefficients used by Maartje for CO2 relperms
257inline double polyC3( double Sw, double C1, double C2, double C3 ) {
258 if ( Sw > 1. ) return 0.;
259 return C1 * std::pow(1. - Sw, C2) * (1. - std::pow(Sw, C3));
260 }
261
262inline double poly_abc( double ux, double a, double b, double c ) {
263 return a * std::pow( ux, b ) + c;
264 }

560struct CRC3_RockType4 {

563 double Krw_ParallelDrainage( double Sw, double ux ) const {
564 if ( Sw <= Swi_ ) return 0.;
565 if ( fabs(Sw - 1.) < std::numeric_limits<double>::epsilon() * 100.) return 1.;
566 if ( ux >= ux_VL_ ) ux = ux_VL_;
567 if ( ux <= ux_CL_ ) ux = ux_CL_;
568 const double rw = r_w( Sw, Swi_ );
569 const double C1_krw = gamma( ux, aw1_, bw1_, cw1_,dw1_ );
570 const double C2_krw = 0.6;
571 const double Krw_ave = k_w( rw, C1_krw, C2_krw );
572 return std::min( std::max( Krw_ave, 0. ), 1. );
573 }

575 double Krn_ParallelDrainage( double Sw, double ux ) const {
576 if ( Sw <= Swi_ ) return 1.;
577 if ( fabs(Sw - 1.) < std::numeric_limits<double>::epsilon() * 100.) return 0.;
578 if ( ux >= ux_VL_ ) ux = ux_VL_;;
579 if ( ux <= ux_CL_ ) ux = ux_CL_;

1308 In the model absolute permeability is a scalar value. We choose this scalar value to be the permeability in the horizontal direction. Anisotropy in the absolute permeability is incorporated through the relative permeability as described below.
1309
1310 Kave_hor = the average cell permeability in the horizontal direction
1311 Kave_ver = the average cell permeability in the vertical direction direction
1312
1313 The water permeability in the vertical direction is given by Kwater=Kave_ver * Krw

1319 KCO2= Kave_hor*(Kave_ver/Kave_hor)* Krnw = Kave_ver * Krnw
