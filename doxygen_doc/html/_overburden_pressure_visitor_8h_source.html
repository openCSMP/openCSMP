20/**
21 For a given Sv (overburden stress at the model top),
22 this visitor computes:
23
24 - the (hydraulic) 'conductivity' from the element 'permeability' and (nodal) 'fluid viscosity'
25
26 - (nodal) 'gravity force' scalar variable as the product of g (specified by user) and the
27 'dry rock density' specified as element property.
28
29 - the (element) 'gravity term' vector variable used in for the integration
30 of the fluid density via the corresponding PDE operator.
31
32 @todo SKM: this visitor should include the influence of pore pressure.
33
34*/
35template<uint32_t dim>
36class OverburdenPressureVisitor : public Visitor<dim> {
37 public:
38 /// for the computation of nodal 'gravity force'
39 OverburdenPressureVisitor( Model<dim>&, double acc_gravity ); ///< local constant for reservoir
40
41 virtual ~OverburdenPressureVisitor();
42
43 /// this method computes all the element properties 'K' and 'gravity term', etc.
44 virtual void Visit( Element<dim>* );
45
46 /// nothing needs to be done at the level of the model
47 virtual void Visit( Model<dim>* ) {}
48
49 private:
50 double dryDensityFromBulkDensity( double rho_bulk, double rho_fluid, double porosity );
51
52 private:
53 const csmp::Index rhof_key_, rhor_key_, ///< (nodal) fluid and dry rock densities (scalars)
54 rhob_key_, phi_key_, ///< bulk- (fluid+rock) density and porosity
55 k_key_, mu_key_, K_key_, ///< (element) permeability, (nodal) 'fluid viscosity' and hydraulic 'conductivity' (scalar)
56 gf_key_, gt_key_; ///< (nodal) gravity force and (elemental) 'gravity term' (vector)
57 const double acc_gravity_;
58 VectorVariable<dim> gravity_;
59};
60
61
62/// calculation ignoring the weight of air.
63template<uint32_t dim>
64inline double OverburdenPressureVisitor<dim>::dryDensityFromBulkDensity( double rho_bulk, double rho_fluid, double porosity )
65 {
66 return rho_bulk - porosity * rho_fluid;
67 }
