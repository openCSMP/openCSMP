121/// establishes 'velocity' and 'volume flux' for a single phase system ('conductivity'), beware of surface elements
122template<uint32_t dim>
123void singlePhaseVelocity( Model<dim>& model, const std::string& regionName,
124 const std::string velocityName = "velocity",
125 const std::string pressureName = "fluid pressure",
126 const std::string conductivityName = "conductivity",
127 const std::string volumeFluxName = "volume flux" )
128 {
129 VectorVariable<dim> velo;
130 ScalarVariable flux;
131 DenseMatrix<DM_MIN> DERIV(dim,dim);
132 Index velocityKey( model.Database().StorageKey( velocityName.c_str() ) );
133 Index fluidPresssureKey( model.Database().StorageKey( pressureName.c_str() ) );
134 Index conductivityKey( model.Database().StorageKey( conductivityName.c_str() ) );
135 Index volumeFluxKey( model.Database().StorageKey( volumeFluxName.c_str() ) );
136
137 Region<dim>& rref( model.Region( regionName.data() ) );
138
139 const auto elementsEnd = rref.CellsEnd();
140 for ( auto it = rref.CellsBegin(); it != elementsEnd; ++it )
141 {
142 // vt = -k (lt grad p)
143 const double conductivity = (*it)->Read( conductivityKey );
144 velo = 0.;
145 (*(*it)).dN_AtBaryCenter( DERIV, 1U );
146 for ( auto i = 0; i < (*it)->Nodes(); ++i )
147 {
148 double pf = (*it)->N(i)->Read( fluidPresssureKey );
149 for( auto xyz = 0; xyz < dim; ++xyz )
150 velo( xyz ) += pf * -DERIV( xyz, i ) * conductivity;
151 }
152 // storing the computed velocity
153 (*it)->Store( velocityKey, velo );
154 (*it)->Store( volumeFluxKey , makeScalar( PLAIN, velo.Length() ) );
155 } // elements region
156 }

196template<uint32_t dim,template<uint32_t> class Domain>
197bool connected( const Domain<dim>& d1, const Domain<dim>& d2 )
198{
199 const typename std::vector<typename Domain<dim>::Simplex*>::const_iterator simplicesEnd = d1.CellsEnd();
200 for ( typename std::vector<typename Domain<dim>::Simplex*>::const_iterator it = d1.CellsBegin(); it != simplicesEnd; ++it )
201 {
202 const typename std::vector<typename Domain<dim>::Simplex*>::const_iterator simplicesEndInner = d2.CellsEnd();
203 for ( typename std::vector<typename Domain<dim>::Simplex*>::const_iterator iit = d2.CellsBegin(); iit != simplicesEndInner; ++iit )
204 for( size_t i(0); i < (*iit)->Neighbors(); ++i )
205 if( (*iit)->Neighbor(i) == (*it) )
206 return true;
207 } // simplices domain 1
208 return false;
209}

212template<uint32_t dim,template<uint32_t> class Domain>
213double averageDistanceBetween( const Domain<dim>& d1, const Domain<dim>& d2 )
214 {
215 double averageDistance( 0. );
216 size_t distanceCount( 0 );
217
218 const typename std::vector<Node<dim>*>::const_iterator nodesEnd = d1.NodesEnd();
219 for ( typename std::vector<Node<dim>*>::const_iterator it = d1.NodesBegin(); it != nodesEnd; ++it )
220 {
221 const typename std::vector<Node<dim>*>::const_iterator nodesEndInner = d2.NodesEnd();
222 for ( typename std::vector<Node<dim>*>::const_iterator iit = d2.NodesBegin(); iit != nodesEndInner; ++iit )
223 {
224 ++distanceCount;
225 averageDistance += (*it)->Coordinate().DistanceTo( (*iit)->Coordinate() );
226 } // nodes domain 2
227 } // nodes domain 1
228
229 return static_cast<double>( averageDistance / distanceCount );
230 }

234template<uint32_t dim,template<uint32_t> class Domain>
235double minimumDistanceBetween( const Domain<dim>& d1, const Domain<dim>& d2 )
236{
237 double minumumDistance( std::numeric_limits<double>::max() );
238
239 const typename std::vector<Node<dim>*>::const_iterator nodesEnd = d1.NodesEnd();
240 for ( typename std::vector<Node<dim>*>::const_iterator it = d1.NodesBegin(); it != nodesEnd; ++it )
241 {
242 const typename std::vector<Node<dim>*>::const_iterator nodesEndInner = d2.NodesEnd();
243 for ( typename std::vector<Node<dim>*>::const_iterator iit = d2.NodesBegin(); iit != nodesEndInner; ++iit )
244 {
245 double const currentDistance = (*it)->Coordinate().DistanceTo( (*iit)->Coordinate() );
246 minumumDistance = currentDistance < minumumDistance ? currentDistance : minumumDistance;
247 }
248 } // nodes domain 1
249
250 return minumumDistance;
