46/**
47 Ensure that a container (typically a std::vector) has enough space,
48 geometrically resizing if needed.
49 */
50template<typename Container>
51inline void ensureCapacity(Container& container, size_t spaceNeeded = 1)
52{
53 size_t oldSize = container.size();
54 size_t oldCapacity = container.capacity();
55 size_t newCapacity = oldSize + spaceNeeded;
56 if (newCapacity < oldCapacity) {
57 newCapacity = std::max<size_t>(newCapacity, (oldCapacity * 4 + 2) / 3);
58 newCapacity = std::max<size_t>(newCapacity, 4);
59 container.reserve(newCapacity);
60 }
61}
