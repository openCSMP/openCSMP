16/**
17 @brief 2-phase multiphase flow functions for immiscible displacement of slightly compressible fluids.
18
19 Generates all relevant input values for 2-phase immiscible (saturation-based volume) transport scheme.
20 Uses the SaturationFunctions and Fluid module and the
21 constitutive relationships specified therein.
22*/
23template<uint32_t dim, template<uint32_t> class USER>
24class TwoPhaseFlowFunctions {
25 public:
26 /// current water saturation initialised inside of the model
27 double Sw( Element<dim>* const e ) const;
28
29 /// lambda parameter: 0 for water, 1 for the non-wetting phase
30 double Mobility( Element<dim>* const, uint32_t phase ) const;
31
32 /// lambda parameter: 0 for water, 1 for the non-wetting phase (using prescribed sw)
33 double Mobility_at( Element<dim>* const, uint32_t phase, double sw ) const;
34
35 /// d lambda_i / dsw
36 double MobilityDerivative( Element<dim>* const, uint32_t phase, bool evaluate_numerically=false ) const;
37
38 double MobilityDerivative_at( Element<dim>* const, uint32_t phase, double sw ) const;
39
40 /// lambda_t: sum of phase mobilities
41 double TotalMobility( Element<dim>* const ) const;
42
43 /// lambda_t: sum of phase mobilities (using prescribed sw)
44 double TotalMobility_at( Element<dim>* const, double sw ) const;
45
46 /// lambda overbar: l1 * l2 / l1 + l2 = mobility product / total mobility also known as G
47 double MobilityProduct( Element<dim>* const ) const;
48
49 /// d lambda overbar / dsw also known as dGds
50 double MobilityProductDerivative( Element<dim>* const, bool evaluate_numerically=false ) const;
51
52 double MobilityProductDerivative_at( Element<dim>* const, double sw ) const;
53
54 /// fractional flow; 0=water, 1=non-wetting phase
55 double f( Element<dim>* const, uint32_t phase ) const;
56
57 /// fractional flow; 0=water, 1=non-wetting phase (using prescribed sw)
58 double f_at( Element<dim>* const, uint32_t phase, double sw ) const;
59
60 /// Permeability
61 double Permeability( Element<dim>* const ) const;
62
63 /// derivative of fractional flow (used in advection multiplier); note that fw+fn=1, dfw_dsw=dfn_dsn
64 double dfds( Element<dim>* const, uint32_t phase ) const;
65
66 double dfds_at( Element<dim>* const, double sw ) const;
67
68 /// maximum value of previous derivative
69 double MaxFractionalFlowDerivative( Element<dim>* const ) const;
70
71 /// multiplier for advection viscosity coefficient in the case of non-linear advection
72 double AdvectionMultiplier( Element<dim>* const ) const;
73
74 /// outputs the shock wave celerity
75 double ShockSpeed( Element<dim>* const ) const;
76
77 /// outputs the water saturation at the non-wetting phase shock front
78 double ShockHeight( Element<dim>* const ) const;
79
80 /// calculated shock height and speed
81 void ShockSpeedAndHeight( Element<dim>* const, double& speed, double& height ) const;
82
83 /// multipliers for gravity-driven flow (advection multiplier and source term)
84 void GravityMultiplier( Element<dim>* const, VectorVariable<dim>& dip_vec ) const;
85
86 void GravityMultiplier_G( Element<dim>* const, VectorVariable<dim>& dip_vec ) const;
87
88 void GravityMultiplier_dGds( Element<dim>* const, VectorVariable<dim>& dip_vec ) const;
89
90 /// multiplier for diffusion coefficient in the case of non-linear diffusion uses viscosity(phase)
91 double DiffusionMultiplier( Element<dim>* const, size_t phase ) const;
92
93 /// driven by capillary pressure gradient
94 double CapillaryDiffusionMultiplier( Element<dim>* const ) const;
95
96 double CapillaryDiffusionMultiplier_Phase( Element<dim>* const, size_t phase ) const;
97
98 /// calculating the Shock velocity
99 double ShockFrontVelocity( Element<dim>* const ) const ;
100
101 double dfds_Numerical( Element<dim>* const, uint32_t phase, double delta_s = 0.001 ) const;
102
103 double dfds_at_Numerical( Element<dim>* const, double sw, double delta_s = 0.001 ) const;
104
105 double dGds_Numerical( Element<dim>* const, double delta_s = 0.000001 ) const;
106
107 double dlwds_Numerical( Element<dim>* const, double delta_s = 0.001 ) const;
108
109 double dlnds_Numerical( Element<dim>* const, double delta_s = 0.001 ) const;
110
111
112 private:
113 USER<dim>* User() { return static_cast<USER<dim>*>(this); }
114 USER<dim> const* User() const { return static_cast<const USER<dim>*>(this); }
115
116 // NON-STANDARD INTERFACES
117
118 /// calculating the Inflection point
119 double InflectionPointSaturation( Element<dim>* const ) const ;
120
121 /// calculating the Tangent Point
122 double TangentPointSaturation( Element<dim>* const ) const ;
123
124 /// calculating Buckley Leverett function... this has to be zero at shock point
125 double TangentOfFractionalFlowFunction( Element<dim>* const, double S) const ;
126
127 /// finding root of Buckley Leverett function... which results is shock point saturation.
128 double FindRootSecantMethod( Element<dim>* const, double S1, double S2) const ;
129};

131} // end csmp
