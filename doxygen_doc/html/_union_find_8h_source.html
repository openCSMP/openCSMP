9/**
10 Union-find algorithm. This data structure takes connections between two
11 elements, and finds all of the connected components and their sizes. A
12 "component" is represented by a single representative item.
13
14 See:
15
16 Tarjan, Robert Endre (1975). "Efficiency of a Good But Not Linear
17 Set Union Algorithm". JACM. 22 (2): 215–225.
18 */
19template<typename Item>
20class UnionFind {
21 public:
22 // This code assumes that passing around and storing an Item is very cheap.
23 // Violating this could cause performance issues.
24 static_assert(sizeof(Item) <= 2 * sizeof(size_t), "Item may be too large");
25
26 /// Say that two items are in the same subset.
27 void SameComponent(Item x, Item y);
28
29 /// Extract the components and their sizes.
30 template<typename Container>
31 void Components( Container& container ) const
32 {
33 for (auto& component : components_) {
34 auto& rec = records_[component];
35 container.push_back(std::make_pair(rec.size_, rec.item_));
36 }
37 }
38
39 /// Resolve an item to its representative
40 Item resolve(Item item) {
41 return records_[find_root(ensure(item))].item_;
42 }
43
44 private:
45 struct Record {
46 Item item_;
47 size_t parent_;
48 size_t size_;
49
50 Record(Item item, size_t n)
51 : item_(item), parent_(n), size_(1)
52 {
53 }
54 };
55
56 std::map<Item,size_t> item_map_;
57 std::deque<Record> records_;
58 std::set<size_t> components_;
59
60 // Find the representative node of an item's component by traversing
61 // the tree to the root, compressing the path by halves as we traverse.
62 size_t find_root(size_t i) {
63 while (records_[i].parent_ != i) {
64 size_t& parent = records_[i].parent_;
65 parent = records_[parent].parent_;
66 i = parent;
67 }
68 return i;
69 }
70
71 // Ensure that an item is in the set.
72 size_t ensure(Item item) {
73 size_t id = records_.size();
74 auto result = item_map_.insert(std::make_pair(item, records_.size()));
75 if (result.second) {
76 records_.emplace_back(item, id);
77 components_.insert(id);
78 return id;
79 }
80 else {
81 return result.first->second;
82 }
83 }
84
85};
86
87
88template<typename Item>
89void
90UnionFind<Item>::SameComponent(Item x, Item y)
91{
92 size_t xr = find_root(ensure(x));
93 size_t yr = find_root(ensure(y));
94
95 if (xr == yr) {
96 // The two iterms are already in the same component.
97 return;
98 }
99
100 // Add the smaller component to the larger component.
101 if (records_[xr].size_ < records_[yr].size_) {
102 records_[xr].parent_ = yr;
103 records_[yr].size_ += records_[xr].size_;
104 components_.erase(xr);
105 }
106 else {
107 records_[yr].parent_ = xr;
108 records_[xr].size_ += records_[yr].size_;
109 components_.erase(yr);
110 }
111}
112
113} // csmp
