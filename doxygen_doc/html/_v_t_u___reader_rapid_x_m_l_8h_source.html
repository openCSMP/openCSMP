46/// Compile-time version of member Size(), given space dimension of the model, returns variable values that are stored in the template parameter variable.
47template<uint32_t dim,template<uint32_t> class csmp_var>
48int valuesInType( int components = -1 ) {
49 if constexpr ( std::is_same<csmp_var<dim>,csmp::ScalarVariable>::value_type ) return 1;
50 else if constexpr ( std::is_same<csmp_var<dim>,csmp::VectorVariable<dim>>::value_type ) return dim;
51 else if constexpr ( std::is_same<csmp_var<dim>,csmp::TensorVariable<dim>>::value_type ) return dim * dim;
52 // Array and flagged array need a different function because they cannot be evaluated at compile time
53 else if ( components >= dim ) return components;
54 return csmp::UNSPECIFIED;
55 }
56
57/// Compile-time version of member Size() returns number of values in ScalarVariable and ArrayVariable types.
58template<typename csmp_var>
59int valuesInType( const csmp_var& var = csmp_var{} ) {
60 if constexpr ( std::is_same<csmp_var,csmp::ScalarVariable>::value_type ) return 1;
61 else if constexpr ( std::is_same<csmp_var,csmp::ArrayVariable>::value_type ) return var.Size();
62 else if constexpr ( std::is_same<csmp_var,csmp::FlaggedArrayVariable>::value_type ) return var.Size();
63 else if constexpr ( std::is_same<csmp_var,double>::value_type ) return 1;
64 else if constexpr ( std::is_same<csmp_var,int>::value_type ) return 1;
65 return csmp::UNSPECIFIED;
66 }

68/// Variable type deduction from number of components; functions does does return flagged array type.
69template<uint32_t dim>
70csmp::VARIABLE_TYPE inferTypeFromNumberOfComponents( int components ) {
71 static_assert( dim != 1U, "inferFromNumberOfComponents: impossible task in 1D");
72 if ( components == 1 ) return csmp::SCALAR;
73 else if ( components == dim ) return csmp::VECTOR;
74 else if ( components == dim * dim ) return csmp::TENSOR;
75 else if ( components > dim ) return csmp::ARRAY;
76 std::cerr <<"\n\n"<<"ERROR, inferTypeFromNumberOfComponents("<< components <<"): failed to identify csmp::VARIABLE_TYPE, returning ARRAY"<< std::endl;
77 return csmp::ARRAY;
78 }
