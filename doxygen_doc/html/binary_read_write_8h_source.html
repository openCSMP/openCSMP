155template<class V, class D, uint32_t dim>
156bool variablesOut( std::fstream& fp, const D& domain, const PropertyDatabase<dim>& pref, VARIABLE_TYPE vtype )
157{
158 size_t vcount( pref.VariableCount( domain.Placement(), vtype ) );
159 fp.write( (char*)&vcount, sizeof( size_t ) );
160 std::set<std::string> propList;
161 pref.ListVariables( domain.Placement(), vtype, propList );
162 for ( std::set<std::string>::const_iterator it( propList.begin() ); it != propList.end(); ++it )
163 {
164 V var;
165 Index key( pref.StorageKey( it->c_str() ) );
166 femDataOutputDispatch::initVariable( key, var );
167 domain.Read( key, var );
168 binaryFileWrite( fp, it->c_str() );
169 if ( !var.Out( fp ) )
170 return false;
171 }
172 return true;
173}

299'unsigned long' number which determines the number of objects which are
300written to file.
301
302@section arguments Input Arguments
303
304A file pointer of a binary file opened in write mode (e.g., "wb") and a

306
307@return returns a boolean indicating whether all data have been

309

311

314@section application Application
315

322template<class T>
323bool binaryFileWrite( std::fstream& fp, const std::vector<T>& stl_ctner )
324{
325 const size_t bytes = sizeof(T);
326 const size_t elements(stl_ctner.size());
327
328 if ( stl_ctner.empty() ) {
329 std::cerr <<"\nbinaryFileWrite(vector): WARNING: container is empty."<< std::endl;
330 fp.write( reinterpret_cast<const char*>(&elements), sizeof(size_t) );
331 return false;
332 }
333 if (!fp.is_open()) {
334 std::cerr << "\nbool binaryFileWrite(vector): ERROR: invalid file pointer." << std::endl;
335 return false;
336 }
337
338 // writing the size of the object
339 fp.write( reinterpret_cast<const char*>(&elements), sizeof(size_t) );
340
341 // writing all elements
342 if ( elements >= 1 )
343// fp.write( reinterpret_cast<const char*>(&stl_ctner[0]), bytes * elements );
344 fp.write( reinterpret_cast<const char*>(stl_ctner.data()), bytes * elements );
345
346 return true;
