38// enum to string
39inline std::string parse( LICENSE_PERIOD p )
40 {
41 if ( p == UNLIMITED ) return "UNLIMITED";
42 if ( p == TRIAL ) return "TRIAL";
43 if ( p == YEAR ) return "ONE-YEAR";
44 if ( p == THREE_YEAR ) return "THREE-YEAR";
45 return "'invalid'";
46 }

50// initialisation
51inline Expiration::Expiration( LICENSE_PERIOD p, int startYear, int startMonth, int startDay )
52 : duration(p)
53 {
54 // establishing a time-record for the start date of the license
55 // ------------------------------------------------------------
56 start_time.tm_sec=0.; /* seconds after the minute [0-60] */
57 start_time.tm_min=0.; /* minutes after the hour [0-59] */
58 start_time.tm_hour=0.; /* hours since midnight [0-23] = midnight */
59 start_time.tm_mday=startDay; /* day of the month [1-31] */
60 start_time.tm_mon=startMonth; /* months since January [0-11] */
61 start_time.tm_year=startYear-1900; /* years since 1900 */
62
63 // establishing how many seconds still remain
64 std::time_t now = std::time(NULL);
65
66 #ifdef _MSC_VER
67 #if _MSC_VER < 1800 // detecting versions older than 2013
68 secs_remaining = static_cast<long>(duration) * 86400L - static_cast<long>(ceil(std::difftime( now, std::mktime(&start_time))) );
69 #else
70 secs_remaining = static_cast<long>(duration) * 86400L - lrint(std::difftime( now, std::mktime(&start_time)) );
71 #endif
72 #else
73 secs_remaining = static_cast<long>(duration) * 86400L - lrint(std::difftime( now, std::mktime(&start_time)) );
74 #endif
75// std::cout <<"\nExpiration: remaining time: "<< secs_remaining <<"\n";
76 }
