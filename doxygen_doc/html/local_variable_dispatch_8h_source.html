188 /**
189 Specialization for Element class.
190
191 Offset of sector or facet integration variable records from beginning of data vector in local variable storage associated with an element
192
193 Function used only if index refers to a sector- or facet integration point variable.
194
195 Method takes offset calculated in property database ignoring the number of integration points
196 because these will vary from element to element type.
197 This offset is now expanded for the element variable of interest
198 by multiplicating those other element integration point variables which precede the current
199 variable.
200
201 @attention storage order is: all sector integration point variables for the first ip, then the second
202 and so forth. Subsequently all facet integration point variables in the same fashion.
203
204 @attention: this method is only used for the reading of variables that are stored in the LVS
205 associated with Element objects
206
207 @attention Index::offsetFactorSector is zero for sector integration point variables and 1 for facet integration point variables
208 */
209 template<uint32_t dim>
210 inline std::pair<int_type,int_type> containerOffset( const csmp::Element<dim>* e, const csmp::Index& idx )
211 {
212 const int_type dataOffset(idx.dataOffset
213 + (idx.ipFactorCell + idx.ipFactorSector + idx.ipFactorFacet) * idx.localVariables.totalDataDepth
214 + idx.offsetFactorCell * e->IntegrationPoints() * idx.integrationPointVariables.ipvCell.totalDataDepth
215 + idx.offsetFactorSector * e->IntegrationPointsPerSector() * e->Sectors() * idx.integrationPointVariables.ipvSector.totalDataDepth);
216
217 const int_type flagOffset(idx.flagOffset
218 + (idx.ipFactorCell + idx.ipFactorSector + idx.ipFactorFacet) * idx.localVariables.totalFlagDepth
219 + idx.offsetFactorCell * e->IntegrationPoints() * idx.integrationPointVariables.ipvCell.totalFlagDepth
220 + idx.offsetFactorSector * e->IntegrationPointsPerSector() * e->Sectors() * idx.integrationPointVariables.ipvSector.totalFlagDepth);
221
222 return std::make_pair(dataOffset, flagOffset);
223 }
