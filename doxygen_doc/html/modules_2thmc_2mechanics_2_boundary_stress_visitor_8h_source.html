25/**
26 @brief BoundaryStressVisitor accesses a Boundary object that must be a surface,
27 computing nodal forces based on the normal and shear stresses applied to that Boundary
28 element. If the boundary is inclined or vertical and body forces are enabled,
29 the visitor takes the vertical variation in the boundary normal stress into account.
30
31 @attention this visitor assumes that the driving stresses are Andersonian, i.e.
32 that one of the principal stress axes is aligned with the vertical (Y)-axis of the model.
33
34 @note This visitor allows for a workaround for applying stress boundary conditions
35 if one does not want to use the Neumann condition functionality of the PDE_Integrator.
36
37 Input are the Face variables 'normal stress' and 'shear stress.' The nodal
38 forces are computed from these.
39
40 Use this, for instance to compute and apply overburden stresses.
41
42 @attention by default all forces that were previously assigned are
43 zeroed first.
44
45 @attention compressive (inward pointing) stresses are positive.
46
47 @attention this visitor assumes that the far-field stress regime is
48 Andersonian, i.e. one of the principal stresses is oriented perpendicular
49 to the earth surface.
50
51 @author SKM 29/9/2014
52*/
53template<uint32_t dim>
54class BoundaryStressVisitor : public Visitor<dim> {
55 public:
56 /// by default, any entries into the RHS force vectors will get overwritten
57 BoundaryStressVisitor( const Model<dim>&,
58 const char* SV_variable, ///< should vary with depth
59 bool overwrite_force_vector=true );
60
61 /// computations using absolute stress values on the boundary
62 BoundaryStressVisitor( const Model<dim>&,
63 BOUNDARY_STRESS=BOUNDARY_STRESS::SINGLE_VALUED,
64 bool overwrite_force_vector=true );
65
66 virtual ~BoundaryStressVisitor();
67
68 /// to zero out force vector, prior to accumulation of forces.
69 virtual void Visit( Boundary<dim>* );
70
71 /// computes nodal forces weighted by node
72 virtual void Visit( Face<dim>* );
73
74 private:
75 /// normals at the boundaries are not correct; this is a workaround
76 void RetrieveNormalWithCorrectDirection( Face<dim>*, VectorVariable<dim>& nrml );
77
78 /// no vertical variation
79 void ApplyConstantStressBoundaryConditions( Face<dim>* );
80
81 /// variation with depth due to the action of body force = gravity
82 void ApplyDepthDependentBoundaryConditions( Face<dim>* );
83
84
85 private:
86 const csmp::Index Sv_key_, ///< vertical (overburden) stress (a scalar)
87 SH_max_key_, ///< maximum horizontal stress expressed in terms of SV
88 Sh_min_key_, ///< minimum horizontal stress Sh_min/SV
89 bstress_key_; ///< stress acting on the boundary face (one value per boundary)
90 const csmp::Index F_key_; ///< (nodal) 'force' (N)
91 VectorVariable<dim> vc_, force_; ///< forces read from the nodes
92 VectorVariable<dim> nrml_, bstress_;
93 TensorVariable<dim> stressTensor_;
94 ScalarVariable SHmax_, Shmin_; ///< stress ratios read from the boundary
95 BOUNDARY_STRESS stress_computation_;
96 bool overwrite_previous_forces_;
97};
98
99} // end csmp
