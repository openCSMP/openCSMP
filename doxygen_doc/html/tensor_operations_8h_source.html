41/// tensor component acting normal to a plane: (ts . n) n (Cauchy's formula)
42inline double normalTensorProjection( const TensorVariable<3U>& ts, const Point<3U>& un )
43 {
44 // (t . n) n (eqn. 6.49, P&F, p.216)
45 return ts(0,0) * un[0] * un[0] + ts(1,1) * un[1] * un[1] +
46 ts(2,2) * un[2] * un[2] + 2. * ts(0,1) * un[0] * un[1] +
47 2. * ts(1,2) * un[1] * un[2] + 2. * ts(2,0) * un[2] * un[0];
48 }
49
50inline double normalTensorProjection( const TensorVariable<3U>& ts, const VectorVariable<3U>& un )
51 {
52 // (t . n) n (eqn. 6.49, P&F, p.216)
53 return ts(0,0) * un[0] * un[0] + ts(1,1) * un[1] * un[1] +
54 ts(2,2) * un[2] * un[2] + 2. * ts(0,1) * un[0] * un[1] +
55 2. * ts(1,2) * un[1] * un[2] + 2. * ts(2,0) * un[2] * un[0];
56 }
