58/// write guard for vector variables
59template<uint32_t dim, template<uint32_t> class CELL>
60void writeVariableIf( CELL<dim>* ptr,
61 const csmp::Index& idx,
62 const VectorVariable<dim>& var,
63 VARIABLE_FLAG dont_overwrite )
64 {
65 VectorVariable<dim> vc;
66 ptr->Read( idx, vc );
67 for ( auto i{0u}; i<dim; i++ )
68 // the component gets overwritten
69 if ( ptr->Status(idx,i) != dont_overwrite ) {
70 vc.Flag(i) = var.Flag(i);
71 vc(i) = var[i];
72 }
73 ptr->Store( idx, vc );
74 } // end version for vector variables
